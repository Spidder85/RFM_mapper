<FILTER>
<COMMON>
    <REAS>
        <INC>1</INC>
        <LIST>40</LIST>
    </REAS>
</COMMON>
<CHECKKIND>
    <INC>1</INC>
    <LIST>1;2</LIST>
</CHECKKIND>
<ACTION>
    <INC>1</INC>
    <LIST>1;2</LIST>
</ACTION>
<FILTER_VERSION>43</FILTER_VERSION>
<FILTER_SP>1</FILTER_SP>
</FILTER>
